// a test fixture, not a substitute implementation of production persistence.

	}

	db := g.DB()

			must(e)

			}
		}
	}
	// This invokes the real production entry, which must return before any profile,
	// dispatch, enrichment, asset preparation or cache write for a missing pair.
	_, err := s.commitCollectMaterial(ctx, event(ctx, 20904504), &collectContentResult{}, gdb.Record{}, "fixture")

	// bounded test sink. No production profile/schema/services are simulated.

			for _, v := range verifies {
				ready, e := s.collectEventMediaReady(c, v["id"].Int64())
				if e != nil {
					return e
				}
				if !ready {
					return newCollectProcessRetryError(time.Second, "waiting persisted video")
				}
			}
			rows, e := s.collectEventMediaRows(c, display["id"].Int64())
			if e != nil {
				return e
			}
			content, e := s.mergeCollectMaterialContent(c, display, &collectContentResult{Media: collectMediaRowsToItems(rows, "display")})

				return e
			}
			if e = validateCollectMaterialMedia(content); e != nil {
				return e
			}
			if _, e = tx.Exec(`INSERT INTO admission_fixture_profile(event_id,verify_count) VALUES(?,?) ON CONFLICT(event_id) DO NOTHING`, canonical.Int64(), len(verifies)); e != nil {
				return e
			}
			_, e = tx.Exec(`INSERT INTO admission_fixture_dispatch(event_id,profile_id) SELECT event_id,id FROM admission_fixture_profile WHERE event_id=? ON CONFLICT(event_id) DO NOTHING`, canonical.Int64())
			return e
		})
	}
	if e := admit(20906683); !isCollectProcessRetryError(e) {
		panic(fmt.Sprintf("duplicate must wait: %v", e))
	}
	counts(0)
	fmt.Println("PASS PG duplicate_alias_without_verify_cannot_create")
	// Pair is committed, but the referenced video row has not been persisted.
	_, err = db.Exec(ctx, "UPDATE hg_youban_publish_collect_event SET material_parent_event_id=20904504 WHERE id=20906722")
	must(err)
	if e := admit(20906683); !isCollectProcessRetryError(e) {
		panic(fmt.Sprintf("pair without media must wait: %v", e))
	}
	counts(0)
	missingMediaReady, missingMediaErr := s.ensureCollectPairedVerifyReady(ctx, event(ctx, 20904504))
	must(missingMediaErr)
	if missingMediaReady {
		panic("real ensure admitted pair without persisted media")
	}
	fmt.Println("PASS PG paired_event_without_video_row_no_profile_dispatch")
	// A paired video reference without a prepared asset cannot pass admission.
	_, err = db.Exec(ctx, `INSERT INTO hg_youban_publish_collect_event_media(id,event_id,media_type,source_file_id,cache_status) VALUES(3,20906722,'video','gotd:fixture','pending')`)
	must(err)
	if e := admit(20906683); !isCollectProcessRetryError(e) {
		panic(fmt.Sprintf("pending video must wait: %v", e))
	}
	counts(0)
	pendingDisplay := event(ctx, 20904504)
	pendingReady, pendingErr := s.ensureCollectPairedVerifyReady(ctx, pendingDisplay)
	must(pendingErr)
	if pendingReady {
		panic("real ensure admitted pending video")
	}
	_, pendingErr = s.prepareCollectMaterial(ctx, pendingDisplay, &collectContentResult{})
	if !isCollectProcessRetryError(pendingErr) {
		panic(fmt.Sprintf("real prepare must retry pending video: %v", pendingErr))
	}
	_, pendingErr = s.commitCollectMaterial(ctx, pendingDisplay, &collectContentResult{}, gdb.Record{}, "fixture")
	if !isCollectProcessRetryError(pendingErr) {
		panic(fmt.Sprintf("real commit must retry pending video: %v", pendingErr))
	}
	if len(admissionFixtureEnqueued) != 3 {
		panic("pending video should request one enqueue per call")
	}
	for _, id := range admissionFixtureEnqueued {
		if id != 20906722 {
			panic("wrong media enqueue event")
		}
	}
	for _, table := range []string{"hg_content_profile", "hg_youban_publish_collect_dispatch"} {
		value, e := db.GetValue(ctx, "SELECT count(*) FROM "+table)
		must(e)
		if value.Int() != 0 {
			panic("pending real entry wrote " + table)
		}
	}
	fmt.Println("PASS PG real_ensure_prepare_commit_pending_video_retries_no_business_writes")
	_, err = db.Exec(ctx, `DELETE FROM hg_youban_publish_collect_event_media WHERE id=3`)
	must(err)
	fmt.Println("PASS PG paired_video_cache_pending_no_profile_dispatch")
	// An uncommitted video must be invisible to a separate DB transaction.
	uncommitted := errors.New("rollback video fixture")
	err = db.Transaction(ctx, func(c context.Context, tx gdb.TX) error {
		_, e := tx.Exec(`INSERT INTO hg_youban_publish_collect_event_media(id,event_id,media_type,file_url,cache_status) VALUES(3,20906722,'video','https://fixture.invalid/verify.mp4','ready')`)
		if e != nil {
			return e
		}
		// admit uses the outer context, so it opens a separate transaction/connection.
		if e = admit(20906683); !isCollectProcessRetryError(e) {
			return fmt.Errorf("uncommitted video visible or unexpected failure: %v", e)

		panic(err)
	}

	// group. The fixture mapping alone must never invent a canonical pairing.
	_, err = db.Exec(ctx, `UPDATE hg_youban_publish_collect_event SET material_parent_event_id=20906683 WHERE id=20906722`)
	must(err)
	if e := admit(20906683); !isCollectProcessRetryError(e) {

// Exercise real source/target eligibility, next-delay and outbox SQL. No mocks
// replace these functions; all mutations below belong to the disposable DB.
func runCollectWaitingSchedulePostgres(ctx context.Context) {
	db := g.DB()
	s := &sSysPublish{}
	must := func(err error) {
		if err != nil {
			panic(err)
		}
	}
	exec := func(sql string, args ...any) { _, e := db.Exec(ctx, sql, args...); must(e) }
	exec(`CREATE TABLE hg_youban_publish_collect_event_log(id bigserial PRIMARY KEY,tenant_id bigint,account_id bigint,event_id bigint,stage text,status text,message text,meta_text text,created_at timestamp)`)
	exec(`CREATE TABLE hg_youban_publish_collect_source(id bigint PRIMARY KEY,tenant_id bigint,account_id bigint,channel_blacklisted int DEFAULT 0)`)
	exec(`INSERT INTO hg_youban_publish_collect_source(id,tenant_id,account_id) VALUES(999,78,615)`)
	exec(`CREATE TABLE hg_youban_publish_collect_process_outbox(id bigserial PRIMARY KEY,event_id bigint UNIQUE,tenant_id bigint,account_id bigint,source_id bigint,source_chat_id text,status text,available_at timestamp,created_at timestamp,updated_at timestamp,lease_owner text DEFAULT '',lease_expires_at timestamp,last_error text DEFAULT '',history boolean,attempt_count int DEFAULT 0,delivered_at timestamp)`)
	exec(`INSERT INTO hg_youban_publish_collect_event(id,source_id,source_chat_id,source_message_id,material_role,material_group_status,created_at,received_at,updated_at) VALUES(91001,999,'999001',100,'pending','waiting_verify',now()-interval '2 hours',now()-interval '2 hours',now()-interval '20 seconds'),(91002,999,'999002',200,'pending','waiting_verify',now()-interval '1 minute',now()-interval '1 minute',now()-interval '5 seconds')`)
	payload := collectProcessQueuePayload{TenantId: 78, AccountId: 615, SourceId: 999, SourceChatId: "999001", EventId: 91001}
	getEvent := func(id int64) gdb.Record {
		r, e := db.GetOne(ctx, "SELECT * FROM hg_youban_publish_collect_event WHERE id=?", id)
		must(e)
		return r
	}
	getOutbox := func() gdb.Record {
		r, e := db.GetOne(ctx, "SELECT *,available_at::text AS exact_deadline FROM hg_youban_publish_collect_process_outbox WHERE event_id=?", payload.EventId)
		must(e)
		return r
	}
	sourceRows := func(p collectProcessQueuePayload) gdb.Result {
		m := collectProcessEventModel(ctx, p).Fields("e.*").WhereIn("e.status", collectProcessWindowStatuses())
		rows, e := collectMaterialWindowDueModel(m, time.Now()).All()
		must(e)
		return rows
	}
	check := func(p collectProcessQueuePayload, wantRows int, minDelay, maxDelay time.Duration) {
		target := getEvent(p.EventId)
		rows, _, e := s.loadTargetCollectMaterialWindow(ctx, p, target, target["received_at"].GTime())
		must(e)
		if len(rows) != 1 || rows[0]["id"].Int64() != p.EventId {
			panic(fmt.Sprintf("explicit target anchor lost: event=%d rows=%d", p.EventId, len(rows)))
		}
		rows = sourceRows(p)
		if len(rows) != wantRows {
			panic(fmt.Sprintf("source %d rows=%d want=%d", p.EventId, len(rows), wantRows))
		}
		delay, pending, e := nextCollectProcessDelay(ctx, p)
		must(e)
		if !pending || delay < minDelay || delay > maxDelay {
			panic(fmt.Sprintf("event %d next=%s pending=%v want [%s,%s]", p.EventId, delay, pending, minDelay, maxDelay))
		}
	}
	if collectMaterialUnpairedRetryDelay(getEvent(91001), time.Now()) != collectMaterialVerifyRetryDelay {
		panic("expired retry must be one minute")
	}
	if collectMaterialUnpairedRetryDelay(getEvent(91002), time.Now()) != collectMaterialWaitingVerifyRetryDelay {
		panic("fresh retry must be ten seconds")
	}
	check(payload, 0, 35*time.Second, 41*time.Second)
	fresh := payload
	fresh.EventId = 91002
	fresh.SourceChatId = "999002"
	check(fresh, 0, time.Second, 6*time.Second)
	mixed := payload
	mixed.SourceChatId = ""
	mixed.EventId = 0
	delay, pending, e := nextCollectProcessDelay(ctx, mixed)
	must(e)
	if !pending || delay < time.Second || delay > 6*time.Second {
		panic(fmt.Sprintf("mixed cohorts hid fresh event: %s", delay))
	}
	// Unknown ingestion time follows the short cohort in both SQL and Go,
	// regardless of the historical Telegram timestamp.
	exec(`UPDATE hg_youban_publish_collect_event SET created_at=NULL WHERE id=91002`)
	if collectMaterialUnpairedRetryDelay(getEvent(91002), time.Now()) != collectMaterialWaitingVerifyRetryDelay {
		panic("null ingestion timestamp must match SQL short cohort")
	}
	check(fresh, 0, time.Second, 6*time.Second)
	exec(`UPDATE hg_youban_publish_collect_event SET created_at=now()-interval '1 minute' WHERE id=91002`)
	fmt.Println("PASS PG expired_60s_fresh_10s_source_target_nextdelay_agree")
	exec(`UPDATE hg_youban_publish_collect_event SET updated_at=now()-interval '61 seconds' WHERE id=91001`)
	check(payload, 1, 0, 0)
	exec(`UPDATE hg_youban_publish_collect_event SET updated_at=now()-interval '11 seconds' WHERE id=91002`)
	check(fresh, 1, 0, 0)
	// Real waiting update must refresh the timestamp, not leave a due source
	// spinning even though the event worker requested a long delayed retry.
	must(s.markCollectMaterialWaitingVerify(ctx, 91001))
	check(payload, 0, 55*time.Second, 61*time.Second)
	fmt.Println("PASS PG expired_due_and_waiting_refresh_preserve_backoff")
	// Establish a pending late deadline, then explicitly wake this exact event.
	must(ensureCollectProcessOutbox(ctx, payload, time.Minute))
	late := getOutbox()["exact_deadline"].String()
	must(s.scheduleCollectDisplaySettlement(ctx, payload, getEvent(91001), 0))
	early := getOutbox()["exact_deadline"].String()
	if early >= late {
		panic("explicit wake did not advance deadline")
	}
	worker := payload
	worker.OutboxOwner = "obsolete-worker"
	must(ensureCollectProcessOutbox(ctx, worker, time.Minute))
	if getOutbox()["exact_deadline"].String() != early {
		panic("old worker delayed explicit wake")
	}
	must(ensureCollectProcessOutbox(ctx, payload, time.Minute))
	if getOutbox()["exact_deadline"].String() != early {
		panic("generic late retry delayed earlier available_at")
	}
	if getOutbox()["status"].String() != "pending" {
		panic("old retry consumed explicit wake")
	}
	fmt.Println("PASS PG real_explicit_wake_advances_and_late_retry_keeps_earlier_available_at")
	var wg sync.WaitGroup
	errs := make(chan error, 16)
	for i := 0; i < 16; i++ {
		wg.Add(1)
		go func(i int) {
			defer wg.Done()
			d := time.Minute
			if i%2 == 0 {
				d = 10 * time.Second
			}
			errs <- ensureCollectProcessOutbox(ctx, payload, d)
		}(i)
	}
	wg.Wait()
	close(errs)
	for e := range errs {
		must(e)
	}
	if getOutbox()["exact_deadline"].String() != early {
		panic("concurrent retries delayed earlier wake")
	}
	fmt.Println("PASS PG concurrent_retries_cannot_postpone_pending_wake")
	// A running lease's old, consumed deadline must not defeat a legitimate
	// retry; a worker holding a different lease must not replace current work.
	exec(`UPDATE hg_youban_publish_collect_process_outbox SET status='processing',lease_owner='current-worker',available_at=now()-interval '1 hour' WHERE event_id=?`, payload.EventId)
	current := payload
	current.OutboxOwner = "current-worker"
	must(ensureCollectProcessOutbox(ctx, current, time.Minute))
	row := getOutbox()
	if row["available_at"].GTime().Time.Before(time.Now().Add(55 * time.Second)) {
		panic("current lease reused consumed past deadline")
	}
	exec(`UPDATE hg_youban_publish_collect_process_outbox SET status='processing',lease_owner='new-worker' WHERE event_id=?`, payload.EventId)
	before := getOutbox()["exact_deadline"].String()
	must(ensureCollectProcessOutbox(ctx, current, time.Minute))
	row = getOutbox()
	if row["status"].String() != "processing" || row["lease_owner"].String() != "new-worker" || row["exact_deadline"].String() != before {
		panic("obsolete retry replaced new lease")
	}
	exec(`UPDATE hg_youban_publish_collect_process_outbox SET status='delivered',lease_owner='' WHERE event_id=?`, payload.EventId)
	must(ensureCollectProcessOutbox(ctx, current, time.Minute))
	if getOutbox()["status"].String() != "delivered" {
		panic("obsolete retry reopened delivered event")
	}
	must(ensureCollectProcessOutbox(ctx, payload, time.Minute))
	if getOutbox()["status"].String() != "delivered" {
		panic("ordinary producer reopened delivered HEAD event")
	}
	fmt.Println("PASS PG HEAD_ordinary_producer_preserves_delivered_guard")
	fmt.Println("PASS PG current_lease_retries_but_obsolete_lease_cannot_overwrite_or_reopen")
	// Model the persisted result of binding a new verify: paired groups leave
	// the unpaired waiting state and become eligible without waiting one minute.
	exec(`UPDATE hg_youban_publish_collect_event SET material_role='display',material_group_status='paired',updated_at=now() WHERE id=91001`)
	check(payload, 1, 0, 0)
	must(s.scheduleCollectDisplaySettlement(ctx, payload, getEvent(91001), 0))
	if getOutbox()["status"].String() != "pending" {
		panic("explicit wake cannot reopen delivered event")
	}
	fmt.Println("PASS PG persisted_pair_bypasses_backoff_and_real_wake_reopens_event")
}
